/*  vect0.cpp
    Compilador: bcc32 5.5

    C++ / FERREIRAS/ Arreglos inteligentes: <vector>

  <vector>,  Declaracion de vectores, carga de elementos con push_back, 
  size() del arreglo y  acceso a sus elementos con at(indice).

 <vector> provee una forma de arreglo dinamico que se expande tanto como sea
necesario para acomodar elementos adicionales al final. Los arreglos <vector> son mas 
robustos que los arreglos ordinarios.

*/

#include <iostream>
#include <vector>
#include <string>

using namespace std;

int main(void){

   // Declaracion de vectores

   vector<float> v1; // Crea un vector vacio
   vector<float> v2(10); // Crea un vector con 10 elementos con valor por defecto 0.0
   vector<float> v3(5, 3.33);// Crea un vector con 5 elementos, cada uno con valor 3.33
   vector<string> v4; // Crea un vector vacio de cadenas  

 // push_back para agregar elementos al final del vector

   v1.push_back(1.23);
   v2.push_back(2.34);
   v3.push_back(103.44);
   v4.push_back("Ferreiras");

 // at(), size()

 //  at(indice)  para accesar  elementos de los vectores
 //  indice en el rango [ 0, size() - 1) ] 

   cout << v1.at(0) << endl;
   cout << v2.at(1) << endl;
   cout << v3.at( v3.size()  - 1) << endl;
   cout << v4.at(0) << endl;
   cout << v2.at(10) << endl;
   return 0;
}

      / recurso erase()

#include <iostream>
#include <vector>
 using namespace std;

int main(void){

  system("cls");
  vector<double> v;

  // CARGAMOS ALGUNOS ELEMENTOS EN v

  v.push_back(1.23);
  v.push_back(12.33);
  v.push_back(13.44);
  v.push_back(14.55);
  v.push_back(15.66);
  v.push_back(16.77);
  v.push_back(17.88);

  // VEMOS EL CONTENIDO DEL VECTOR

  cout << endl << "=> El contenido del <vector> es:\n\n";
  for( unsigned int k(0); k < v.size(); k++ )
    cout << "  " << v[k];

  // BORRAMOS EL 6to. ELEMENTO

  v.erase (v.begin() + 5); // begin() apunta a la posicion 1

  // VEMOS EL CONTENIDO DEL VECTOR

  cout << endl << "\n=> El contenido ahora es: " << endl << endl;
  for( unsigned int k(0); k < v.size(); k++ )
    cout << "  " << v[k];

  // BORRAMOS LOS ELEMENTOS EN EL RANGO [3,4] 
 
  v.erase(v.begin() + 2, v.begin() + 4);

// VEMOS EL CONTENIDO DEL VECTOR

  cout << endl << "\n=> El contenido ahora es: " << endl << endl;
  for( unsigned int k(0); k < v.size(); k++ )
    cout << "  " << v[k];
  cout << endl;
  return 0;
}

/*
   vectclear0a.cpp
   Compilador: bcc32 5.5

   C++/ Ferreiras/  Arreglos inteligentes con C++: <vector>
      / recurso clear()

   El recurso clear() de <vector> remueve todos los elementos en
   un <vector> dado, dejandolo con un size de 0 y liberando la
   memoria asignada. 

   Otros recursos asociados son:

   erase()        Borra elementos
   resize()       Cambia el size
   empty()        Verifica si esta vacio

*/

#include <iostream>
#include <vector>
#include <string>
 using namespace std;

int main(void){

  system("cls"); 
 
  vector<string> v;
  cout << "\n=> La capacidad del <vector> actual es: "
       << v.capacity() << endl;

  v.push_back("Estamos en <vector>, ! que bien !");
  v.push_back("Y tenemos 6 puntos");
  v.push_back("Para el next lab");
  
  cout << "\n=> La capacidad del <vector> actual es: "
       << v.capacity() << endl;

  cout << "\n=> El contenido actual en el <vector> es:\n\n";
  for ( unsigned int i(0); i < v.size(); i++)
     cout << "  " << v[i] << endl; 
  
  v.clear();
  if( v.empty()){
    cout << "\n=> Estoy vacio, mi size es: " 
         << v.size() << "  elementos" << endl;
  }
  else{
    cout << "\n=> Estoy lleno con " << v.size() << " elementos\n"
         << endl << " Los cuales son: " << endl;   
    for( unsigned int k(0); k < v.size(); k++)
       cout << "   " << v[k] << endl;    
  }
  cout << "\n=> La capacidad del <vector> actual es: "
       << v.capacity() << endl;

  v.push_back("Ahora le cargo un elemento");
  v.push_back("Otro elemento");

  cout << "\n=> La capacidad del <vector> actual es: "
       << v.capacity() << endl;

  if( v.empty()){
    cout << "\n=> Estoy vacio, mi size es: " 
         << v.size() << "  elementos" << endl;
  }
  else{
    cout << "\n=> Me han empujado elementos y ahora tengo " 
         << v.size() << ", los cuales son: " << endl << endl;   
    for( unsigned int k(0); k < v.size(); k++)
       cout << "   " << v[k] << endl;    
  }
  return 0;
}

  (La captura terminara con <Ctrl>+Z+<ENTER>) y se manipulará dicho vector 
  los recursos indicados. 
 

*/

#include <vector>
#include <iostream>
#include <conio>

using namespace std;

int main(void){

  clrscr();
  cout << endl << endl;

  vector<float> v1;  
  float real;
 
 // max_size()

  cout << "La cantidad maxima de elementos que este vector puede almacenar es:\n "
          <<  v1.max_size() << " Elementos de tipo float, !! Ei diablo !!"
          << endl  << "\nCARGA DE ELEMENTOS EN EL ARREGLO\n"
          << "ENTRE VALORES REALES. PARA TERMINAR CTRL+Z DOS VECES:\n" << endl;
  while( cin >> real) {
     v1.push_back(real);
  }

// size()
 
  cout << endl << "La cantidad actual de elementos en el vector es: "
          << v1.size()  << "\nLos cuales son: " << endl << endl;

  // at(),  back(),  front()

  //  at(i)  Retorna el elemento indice i
  // back()  Retorna el ultimo elemento en el vector 
  // front() Retorna el primer elemento en el vector

  unsigned int j(0);
  while( j < v1.size() ){ 
    cout << v1.at(j) << ",  ";
    j++;
  }
  
  if( !v1.empty() )
      cout << endl << endl << "El elemento (i= 0, n = 1) es at(0) = " << v1.at(0) 
           << endl << endl << "El primer elemento en el vector es: " << v1.front()
           << endl << endl << "El ultimo elemento en el vector es: " << v1.back()
           << endl << endl;
  else
      cout << "\n\nEL VECTOR ESTA VACIO !!" << endl;           

  return 0;
}

/*
  vect2D1b.cpp
  Compilador: bcc32 5.5

  C++/ Ferreiras/ <vector> / 2D

  Para trabajar con matrices:

  (1)    vector<int, allocator<int> >

  (2)   usar la librería Matrix en www.boost.org 

  (3)  Usar un vector de vectores
 
       vector< vector<T> > id_matr; // deje los espacios indicados

  (4) Usar

      typedef vector <T> columnas;
      vector <columnas> filas;

  (5) Tanto en 3 como en 4, despues se deberán crear las filas
      inyectandole vectores con push_back, previamente cargados.
      Cada vez que inyectamos un vector es una fila nueva en la 
      matriz. Por ejemplo:

      vector < vector<int> > matr;  
      matr.push_back(vector <int>()); 

      Creara la fila matr[0]
     
      Podemos asignarle valores, por ejemplo, a matr[0][0]:
    
      matr[0].push_back(50); 
   
      En el siguiente codigo ilustramos esto mas ampliamente.

      !! Cada fila de la matriz podra tener la cantidad de elementos
      que el usuario quiera !!

*/

#include <vector>
#include <iostream>

using namespace std;

int main(void){

    vector< vector<int> > matr;
 // CARGA DE LA MATRIZ
 // CADA VEZ QUE INYECTEMOS UN VECTOR A LA MATRIZ, ES UNA FILA NUEVA

    vector<int> x;
    int y;
    char resp;

    do{
       cout << "\nEntre los elementos de las filas" << endl;
       x.clear(); // borra los elementos en el vector, si los hay.
        // Si el vector no es clear(), simplemente se van agregando
        // dinamicamente y los elementos de las filas anterior
        // estarán en la fila siguiente. (\_~)
       
       while(cin >> y)
         x.push_back(y); // carga los elementos de la fila
       matr.push_back(x); //inyectamos el vector o fila nueva
       cin.clear(); // limpia el buffer 
       cout << "\nOTRA FILA ? s/n: ";
       cin >> resp;
    }while(resp == 's' || resp == 'S');

 // PROCESAMOS LOS ELEMENTOS DE LA MATRIZ, POR EJEMPLO, MOSTRARLOS
               
   cout << "\n\nCantidad de filas = " << matr.size() << endl;
   for(unsigned f(0); f < matr.size(); f++) {
     cout << "\nCantidad de columnas en la fila " << f 
          << " es " << matr.at(f).size() << endl
          << "Estos elementos son: " << endl;
     for (unsigned j(0); j < matr[f].size(); j++) {
        cout << matr[f][j] << " ";
     }
        cout << endl;
   }

  return 0;
}
